## `app.py`

```python
import hashlib
import sqlite3
import subprocess

import yaml
from flask import Flask, request
from markupsafe import Markup

app = Flask(__name__)

# VULN 1 : secret codé en dur (Bandit B105)
SECRET_KEY = "super-secret-password-123"
app.config["SECRET_KEY"] = SECRET_KEY

DB = "users.db"


def init_db():
    conn = sqlite3.connect(DB)
    conn.execute("CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name TEXT, password TEXT)")
    conn.execute("DELETE FROM users")
    # VULN 6 (bis) : MD5 aussi pour le compte admin initial (Bandit B324)
    pwd = hashlib.md5(b"admin123").hexdigest()
    conn.execute("INSERT INTO users (name, password) VALUES ('admin', ?)", (pwd,))
    conn.commit()
    conn.close()


@app.route("/")
def index():
    return "Vulnerable Notes API : /user?name=, /hello?name=, /ping?host=, /config (POST), /register (POST)"


# VULN 2 : injection SQL (Bandit B608)
@app.route("/user")
def get_user():
    name = request.args.get("name", "")
    conn = sqlite3.connect(DB)
    query = "SELECT id, name FROM users WHERE name = '" + name + "'"
    rows = conn.execute(query).fetchall()
    conn.close()
    return {"users": rows}


# VULN 3 : XSS réfléchi (Bandit B704 : Markup sur une donnée non fiable)
@app.route("/hello")
def hello():
    name = request.args.get("name", "inconnu")
    return Markup(f"<h1>Bonjour {name}</h1>")


# VULN 4 : injection de commandes (Bandit B602)
@app.route("/ping")
def ping():
    host = request.args.get("host", "127.0.0.1")
    result = subprocess.run("ping -c 1 " + host, shell=True, capture_output=True, text=True)
    return f"<pre>{result.stdout}</pre>"


# VULN 5 : désérialisation YAML non sûre (Bandit B506)
@app.route("/config", methods=["POST"])
def load_config():
    data = yaml.load(request.data, Loader=yaml.Loader)
    return {"config": str(data)}


# VULN 6 : hash faible MD5 pour les mots de passe (Bandit B324)
@app.route("/register", methods=["POST"])
def register():
    name = request.form["name"]
    password = request.form["password"]
    hashed = hashlib.md5(password.encode()).hexdigest()
    conn = sqlite3.connect(DB)
    conn.execute("INSERT INTO users (name, password) VALUES (?, ?)", (name, hashed))
    conn.commit()
    conn.close()
    return {"status": "created"}


if __name__ == "__main__":
    init_db()
    # VULN 7 : debug activé (B201) + écoute sur toutes les interfaces (B104)
    app.run(host="0.0.0.0", port=5000, debug=True)
```

SonarQube Community n'a pas toutes les règles d'analyse de flux : il peut ne
pas signaler l'injection SQL ou le XSS (surtout des "security hotspots").
Prévoir ~4 Go de RAM ; sous Linux : `sudo sysctl -w vm.max_map_count=524288`.
```bash

# Docker Desktop (Mac/Windows) : remplacer par
#   -Dsonar.host.url=http://host.docker.internal:9000   (et retirer --network host)
```

## Corrections attendues

| # | Faille | Correction |
|---|---|---|
| 1 | Secret en dur | `os.environ["SECRET_KEY"]` (lancer avec `-e SECRET_KEY=...`) |
| 2 | Injection SQL | `conn.execute("SELECT id, name FROM users WHERE name = ?", (name,))` |
| 3 | XSS | retirer `Markup` et utiliser `markupsafe.escape(name)` |
| 4 | Injection de commandes | `subprocess.run(["ping", "-c", "1", host], shell=False)` + validation avec `ipaddress.ip_address(host)` |
| 5 | YAML non sûr | `yaml.safe_load(request.data)` |
| 6 | MD5 | `werkzeug.security.generate_password_hash`, dans `register()` ET dans `init_db()` |
| 7 | Debug / bind | `debug=False`, hôte lu dans `APP_HOST` (défaut `127.0.0.1`) ; dans Docker : `-e APP_HOST=0.0.0.0` |

Note : après correction, Bandit signale encore B404/B603/B607 (niveau Low) sur
`subprocess`. Ce sont des avertissements de revue, pas des failles : une fois
l'entrée validée, on les justifie avec `# nosec B404` sur l'import et
`# nosec B603 B607` sur l'appel.

## Mise à jour des dépendances
Remplacer le contenu de `requirements.txt` (supprimer aussi les anciennes
lignes `MarkupSafe`, `itsdangerous` et `click`, incompatibles avec Flask 3) :
```
Flask>=3.1.3
Werkzeug>=3.1.9
Jinja2>=3.1.6
PyYAML>=6.0.3
requests>=2.33.0
```
Pourquoi ? Les CVE ne sont corrigées que dans les versions récentes. Ici,
PyYAML 5.3.1 permet l'exécution de code arbitraire via un YAML malveillant,
Werkzeug/Jinja2 exposent à des dénis de service, et requests 2.25.0 peut fuiter
l'en-tête Proxy-Authorization. On ne peut pas « patcher » une bibliothèque
tierce soi-même de façon fiable : mettre à jour est la seule vraie remédiation.

## `solution/app.py` (reference answer)

```python
import ipaddress
import os
import sqlite3
import subprocess  # nosec B404

import yaml
from flask import Flask, request
from markupsafe import escape
from werkzeug.security import generate_password_hash

app = Flask(__name__)

# FIX 1 : le secret vient de l'environnement
app.config["SECRET_KEY"] = os.environ["SECRET_KEY"]

DB = "users.db"


def init_db():
    conn = sqlite3.connect(DB)
    conn.execute("CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name TEXT, password TEXT)")
    conn.execute("DELETE FROM users")
    # FIX 6 (bis) : plus de MD5 pour le compte admin non plus
    pwd = generate_password_hash("admin123")
    conn.execute("INSERT INTO users (name, password) VALUES ('admin', ?)", (pwd,))
    conn.commit()
    conn.close()


@app.route("/")
def index():
    return "Vulnerable Notes API : /user?name=, /hello?name=, /ping?host=, /config (POST), /register (POST)"


# FIX 2 : requête paramétrée
@app.route("/user")
def get_user():
    name = request.args.get("name", "")
    conn = sqlite3.connect(DB)
    rows = conn.execute("SELECT id, name FROM users WHERE name = ?", (name,)).fetchall()
    conn.close()
    return {"users": rows}


# FIX 3 : échappement de la sortie
@app.route("/hello")
def hello():
    name = request.args.get("name", "inconnu")
    return f"<h1>Bonjour {escape(name)}</h1>"


# FIX 4 : validation de l'IP + pas de shell
@app.route("/ping")
def ping():
    host = request.args.get("host", "127.0.0.1")
    try:
        ipaddress.ip_address(host)
    except ValueError:
        return {"error": "adresse IP invalide"}, 400
    result = subprocess.run(["ping", "-c", "1", host], shell=False, capture_output=True, text=True)  # nosec B603 B607
    return f"<pre>{escape(result.stdout)}</pre>"


# FIX 5 : safe_load
@app.route("/config", methods=["POST"])
def load_config():
    data = yaml.safe_load(request.data)
    return {"config": str(data)}


# FIX 6 : hash salé et lent
@app.route("/register", methods=["POST"])
def register():
    name = request.form["name"]
    password = request.form["password"]
    hashed = generate_password_hash(password)
    conn = sqlite3.connect(DB)
    conn.execute("INSERT INTO users (name, password) VALUES (?, ?)", (name, hashed))
    conn.commit()
    conn.close()
    return {"status": "created"}


if __name__ == "__main__":
    init_db()
    # FIX 7 : pas de debug ; hôte d'écoute configurable (127.0.0.1 par défaut)
    app.run(host=os.environ.get("APP_HOST", "127.0.0.1"), port=5000, debug=False)
```
